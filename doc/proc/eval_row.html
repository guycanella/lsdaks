    function eval_row(spl, i_x, y) result(f_row)
        type(spline2d_t), intent(in) :: spl
        integer, intent(in) :: i_x
        real(dp), intent(in) :: y
        real(dp) :: f_row

        integer :: ny

        ny = spl%n_y(i_x)
        f_row = spline1d_eval(spl%y(1:ny, i_x), spl%f(1:ny, i_x), &
                              spl%d2f_dy2(1:ny, i_x), ny - 1, y)
    end function eval_row
