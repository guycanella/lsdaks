    pure function b_guess_for(n_tot, m, u) result(b0)
        real(dp), intent(in) :: n_tot, m, u
        real(dp) :: b0

        b0 = (2.0_dp * u / PI) * log(1.0_dp + n_tot / max(m, 1.0e-300_dp))
        b0 = min(max(b0, 10.0_dp * B_MIN), b_max_for(u))
    end function b_guess_for
