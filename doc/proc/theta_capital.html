    function Theta_capital(x, U) result(res)
        real(dp), intent(in) :: x, U
        real(dp) :: res

        if (abs(U) < U_SMALL) then
            res = PI * sign(1.0_dp, x)
        else
            res = 2.0_dp * atan2(2.0_dp * x, U)
        end if
    end function Theta_capital
