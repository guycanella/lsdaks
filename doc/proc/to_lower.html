    pure function to_lower(str) result(folded)
        character(len=*), intent(in) :: str
        character(len=len(str)) :: folded
        integer :: i, code

        do i = 1, len(str)
            code = iachar(str(i:i))
            if (code >= iachar('A') .and. code <= iachar('Z')) then
                folded(i:i) = achar(code - iachar('A') + iachar('a'))
            else
                folded(i:i) = str(i:i)
            end if
        end do
    end function to_lower
