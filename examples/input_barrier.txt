!> Electrons tunneling through a single rectangular barrier
!!
!! `potential_type` must name one of the implemented generators. This example
!! used to say 'barrier', which does not exist, and the run aborted.
!! The single barrier covers exactly `width` sites centred on `position`
!! (1-indexed), with height V0. For the barrier-well-barrier geometry use
!! potential_type = 'barrier_double' with barrier_width, well_depth, well_width.

&system
  L = 40
  Nup = 15
  Ndown = 15
  U = 2.0
  bc = 'periodic'
/

&potential
  potential_type = 'barrier_single'
  V0 = 5.0                  ! Barrier height
  position = 20             ! Barrier centre (lattice site, 1-indexed)
  width = 4                 ! Barrier width, in sites
/

&scf
  max_iter = 150
  density_tol = 1.0e-6
  mixing_alpha = 0.25
  verbose = .true.
/

&output
  output_prefix = 'quantum_barrier'
  save_density = .true.
  save_eigenvalues = .true.
/
