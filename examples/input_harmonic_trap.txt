!> Atoms in a harmonic trap: V(i) = k*(i - (L+1)/2)^2
!!
!! The trap strength is `spring_constant`, NOT `V0`. This example used to set
!! V0 = 0.1 and the run silently used the default k = 0.001, i.e. a trap 100
!! times shallower than advertised. The trap is always centred on (L+1)/2 by
!! construction (see src/potentials/potential_harmonic.f90), so there is no
!! centre parameter to set.

&system
  L = 30
  Nup = 10
  Ndown = 10
  U = 4.0
  bc = 'open'
/

&potential
  potential_type = 'harmonic'
  spring_constant = 0.02    ! Trap strength k
/

&scf
  max_iter = 200
  density_tol = 1.0e-6
  mixing_alpha = 0.3
  verbose = .true.
/

&output
  output_prefix = 'harmonic_trap'
  save_density = .true.
  save_eigenvalues = .true.
/
